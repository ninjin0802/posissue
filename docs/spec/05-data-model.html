5. データモデル (D1)

CREATE TABLE projects (
  key             TEXT PRIMARY KEY,        -- "pk_" + 24 文字
  installation_id INTEGER NOT NULL,
  repo            TEXT NOT NULL UNIQUE,    -- "owner/name"
  allowed_origins TEXT NOT NULL DEFAULT '', -- カンマ区切り。空なら Origin 検査なし
  daily_limit     INTEGER,                 -- NULL なら既定値
  active          INTEGER NOT NULL DEFAULT 1,   -- App のアンインストール等で 0
  banned          INTEGER NOT NULL DEFAULT 0,   -- 運営による停止。/setup では解除しない
  created_at      TEXT NOT NULL DEFAULT (datetime('now'))
);
CREATE INDEX projects_installation ON projects(installation_id);

CREATE TABLE usage (            -- プロジェクトごとの日次件数
  key   TEXT NOT NULL,
  day   TEXT NOT NULL,          -- "YYYY-MM-DD"（UTC）
  count INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (key, day)
);

CREATE TABLE recent_hashes (    -- 重複投稿検出
  hash       TEXT PRIMARY KEY,  -- sha256(key + title + body)
  created_at INTEGER NOT NULL   -- epoch 秒。24 時間より古いものは無視・削除
);
  • 報告者の IP や UA は保存しない（IP は Rate Limiting バインディングのキーとしてのみ使う）。
  • キーの停止は active = 0。削除はしない。
  • allowed_origins と daily_limit は、作者がキー確認画面の「設定」から変える（4.6）。
